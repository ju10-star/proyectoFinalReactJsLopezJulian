import { useContext } from "react";
import { CartContext } from "../context/CartContext";
import { Link, useLocation } from "react-router-dom";
import "./ProductList.css";

export default function ProductCard({ product, onEdit, onToggleVisibility }) {
  const { addToCart, removeFromCart } = useContext(CartContext);
  const location = useLocation();

  // Detectar si estamos en la vista de admin
  const isAdmin = location.pathname.includes("/admin");

  return (
    <div className="product-card">
      <Link to={`/detalle/${product.id}`}>
        <img
          src={
            product.image && !product.image.startsWith("http")
              ? `/src/media/${product.image}`
              : product.image
          }
          alt={product.name}
          className="product-image"
        />
      </Link>

      <div className="product-info">
        <h3>{product.name}</h3>
        <p className="desc">{product.description}</p>
        <p className="price">${product.price}</p>
      </div>

      <div className="product-actions">
        {/* 🛒 Botones visibles en catálogo (inicio) */}
        {!isAdmin && (
          <>
            <button onClick={() => addToCart(product)}>Añadir al carrito</button>
            <button
              onClick={() => removeFromCart(product.id)}
              className="delete"
            >
              Quitar del carrito
            </button>
          </>
        )}

        {/* ⚙️ Botones visibles solo en /admin */}
        {isAdmin && (
          <>
            <button onClick={() => onEdit(product)} className="edit">
              Editar
            </button>
            <button
              onClick={() => onToggleVisibility(product, !product.is_public)}
              className={product.is_public ? "delete" : "add"}
            >
              {product.is_public
                ? "Quitar del catálogo"
                : "Añadir al catálogo"}
            </button>
          </>
        )}
      </div>
    </div>
  );
}
